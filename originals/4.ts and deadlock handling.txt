
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <semaphore.h>
#include <unistd.h>

#define N 5

sem_t forks[N];
sem_t room;

void *philosopher(void *num)
{
    int id = *(int *)num;

    while(1)
    {
        printf("Philosopher %d is Thinking\n", id);
        sleep(1);

        sem_wait(&room);

        sem_wait(&forks[id]);
        printf("Philosopher %d picked LEFT fork %d\n",
               id, id);

        sem_wait(&forks[(id + 1) % N]);
        printf("Philosopher %d picked RIGHT fork %d\n",
               id, (id + 1) % N);

        printf("Philosopher %d is Eating\n", id);
        sleep(2);

        sem_post(&forks[id]);
        sem_post(&forks[(id + 1) % N]);

        printf("Philosopher %d released forks\n", id);

        sem_post(&room);

        sleep(1);
    }

    pthread_exit(NULL);
}

int main()
{
    pthread_t phil[N];
    int id[N];
    int i;

    sem_init(&room, 0, N - 1);

    for(i = 0; i < N; i++)
        sem_init(&forks[i], 0, 1);

    for(i = 0; i < N; i++)
    {
        id[i] = i;

        pthread_create(&phil[i],
                       NULL,
                       philosopher,
                       &id[i]);
    }

    for(i = 0; i < N; i++)
        pthread_join(phil[i], NULL);

    for(i = 0; i < N; i++)
        sem_destroy(&forks[i]);

    sem_destroy(&room);

    return 0;
}
